\abschnitt{Was ist ein FabLab?}
\stichwort{Offene Werkstatt}
\begin{itemize}
  \item Jede:r kann kommen und eigene Ideen umsetzen
  \item Selbermachen und lernen, wie es geht -- nicht machen lassen
\end{itemize}
\vspace{10mm}
\stichwort{Digitale Fertigung}
\begin{itemize}
  \item computergesteuerte Maschinen wie Lasercutter, 3D-Drucker und
    Fräse
\end{itemize}
\vspace{10mm}
\stichwort{Weltweit vernetzt}
\begin{itemize}
  \item mehr als 1.750 FabLabs in rund 100 Ländern, die Fab-Charta als
    gemeinsame Grundlage
  \item Entwürfe und Wissen miteinander teilen
\end{itemize}
